	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"os"
	"path/filepath"
	"strings"
	"sync/atomic"
	"testing"
	"time"

	"github.com/aiii-dot-id/aii-os/internal/broker"
	"github.com/aiii-dot-id/aii-os/internal/packagefmt/packagetest"
	"github.com/aiii-dot-id/aii-os/internal/untrusted"
)

type jevTransport func(*http.Request) (*http.Response, error)

func (f jevTransport) RoundTrip(r *http.Request) (*http.Response, error) { return f(r) }

func TestGenericJevRealBroker(t *testing.T) {
	root := os.Getenv("JEV_STAGE")
	if root == "" {
		t.Fatal("JEV_STAGE is required")
	}
	manifest, err := os.ReadFile(filepath.Join(root, "manifest.json"))
	if err != nil {
		t.Fatal(err)
	}
	var meta struct {
		ID, Version  string
		Capabilities []string `json:"capability_envelope"`
	}
	if err = json.Unmarshal(manifest, &meta); err != nil {
		t.Fatal(err)
	}
	spec := packagetest.PackageSpec{Root: meta.ID + "-" + meta.Version, Manifest: manifest, InstallFiles: map[string][]byte{}}
	err = filepath.WalkDir(filepath.Join(root, "install-root"), func(p string, d os.DirEntry, e error) error {
		if e != nil {
			return e
		}
		if d.IsDir() {
			return nil
		}
		rel, e := filepath.Rel(filepath.Join(root, "install-root"), p)
		if e != nil {
			return e
		}
		b, e := os.ReadFile(p)
		spec.InstallFiles[filepath.ToSlash(rel)] = b
		return e
	})
	if err != nil {
		t.Fatal(err)
	}
	caps := meta.Capabilities

	if err = signer(t).SignT2(&spec, meta.ID, meta.Version, caps); err != nil {
		t.Fatal(err)
	}
	const secret = "fixture-only-jev-key"
	t.Setenv("JEV_FIXTURE_SECRET", secret)
	var calls atomic.Int32
	var mode atomic.Int32
	entered := make(chan struct{}, 1)
	reply := `{"model":"jev-1.13.0","answers":{"n":{"type":"noul","noul":0.9},"c":{"type":"choice","choice":"a","probabilities":{"a":0.8,"b":0.2},"confidence":0.6},"s":{"type":"score","score":1.5,"legend":{"0":"none","1":"some","2":"all"},"probabilities":{"0":0,"1":0.5,"2":0.5},"confidence":0.4}},"usage":{"input_tokens":100,"output_tokens":20}}`
	transport := jevTransport(func(r *http.Request) (*http.Response, error) {
		calls.Add(1)
		if r.URL.Host != "api.typesafe.ai" || r.Header.Get("Authorization") != "Bearer "+secret {
			t.Error("wrong destination or missing broker credential")
		}
		if mode.Load() == 5 {
			entered <- struct{}{}
			<-r.Context().Done()
			return nil, r.Context().Err()
		}
		status := 200
		body := reply
		if r.URL.Path == "/v1/models" {
			if r.Method != "GET" {
				t.Error("models did not use GET")
			}
			body = `{"models":[{"name":"jev-latest"},{"name":"jev-preview"}]}`
			if mode.Load() == 4 {
				body = `{"models":[{"name":"jev-latest"},{}]}`
			}
		} else {
			if r.Method != "POST" || r.URL.Path != "/v1/systemone" {
				t.Error("unexpected request")
			}
			data, e := io.ReadAll(r.Body)
			if e != nil {
				t.Error(e)
			}
			var sent map[string]json.RawMessage
			if e = json.Unmarshal(data, &sent); e != nil || len(sent) != 3 || sent["state"] == nil || sent["questions"] == nil || string(sent["model"]) != `"jev-latest"` {
				t.Errorf("unexpected outbound fields: %s %v", data, e)
			}
			if strings.Contains(string(data), "_host") {
				t.Error("host metadata escaped")
			}
			switch mode.Load() {
			case 1:
				body = strings.Replace(reply, `"noul":0.9`, `"noul":2`, 1)
			case 2:
				status = 429
				body = `{"error":"busy"}`
			case 3:
				body = `{"padding":"` + strings.Repeat("x", 17000) + `"}`
			}
		}
		return &http.Response{StatusCode: status, Header: http.Header{"Content-Type": []string{"application/json"}}, Body: io.NopCloser(strings.NewReader(body)), Request: r}, nil
	})
	st := newBrokerStore(t)
	grant := map[string]broker.Grant{meta.ID: {Hosts: []string{"api.typesafe.ai:443"}, CredentialHandles: []string{"jev"}}}
	profiles := map[string]broker.AuthProfile{"jev": {SecretEnv: "JEV_FIXTURE_SECRET", Host: "api.typesafe.ai", Port: 443}}
	h, err := broker.New(broker.Config{Store: st, Grants: grant, AuthProfiles: profiles, Guard: func(context.Context, string) error { return nil }, Transport: transport})
	if err != nil {
		t.Fatal(err)
	}
	reg := newRegistry(t)
	values := map[string]interface{}{"api_key": "jev", "model": "invalid model"}
	ap, err := Activate(t.Context(), writePkg(t, spec), reg, &Options{Roots: signerRoots(t), Broker: h, Settings: func(string) map[string]interface{} { return values }})
	if err != nil {
		t.Fatal(err)
	}
	defer ap.Deactivate(context.Background())
	foundChoices := false
	for _, setting := range ap.Settings {
		if setting.Key == "model" && setting.ChoicesFrom == "models" {
			foundChoices = true
		}
	}
	if !foundChoices {
		t.Fatal("packaging lost the model picker operation")
	}
	if len(reg.Search("jev", 10)) == 0 {
		t.Fatal("not discoverable")
	}
	models := ToolNameFor(meta.ID, "models")
	r, err := reg.ExecuteModel(t.Context(), models, map[string]interface{}{})
	if err != nil || r.Error != "" || !strings.Contains(r.Output, "jev-preview") {
		t.Fatalf("models: %+v %v", r, err)
	}
	mode.Store(4)
	r, err = reg.ExecuteModel(t.Context(), models, map[string]interface{}{})
	if err != nil || r.Error == "" {
		t.Fatalf("partial model list accepted: %+v %v", r, err)
	}
	mode.Store(0)
	values["model"] = "jev-latest"
	if meta.ID != "id.aiii.jev" {
		return
	}
	name := ToolNameFor(meta.ID, "query")
	card, err := reg.Show(name)
	if err != nil || card.Effects != "write.external" {
		t.Fatalf("descriptor: %+v %v", card, err)
	}
	var args map[string]interface{}
	if err = json.Unmarshal([]byte(`{"state":"evidence","questions":{"n":{"type":"noul","instructions":"Is it supported?"},"c":{"type":"choice","instructions":"Choose","criteria":{"a":null,"b":null}},"s":{"type":"score","instructions":"Rate","criteria":["none","some","all"]}}}`), &args); err != nil {
		t.Fatal(err)
	}
	r, err = reg.ExecuteModel(t.Context(), name, args)
	if err != nil || r.Error != "" || !strings.Contains(r.Output, `"noul":0.9`) || !strings.Contains(r.Output, `"score":1.5`) || !strings.Contains(r.Output, untrusted.Open) || strings.Contains(r.Output, secret) {
		t.Fatalf("mixed reply: %+v %v", r, err)
	}
	for m, code := range map[int32]string{1: "JEV_RESPONSE_INVALID", 2: "JEV_RATE_LIMITED", 3: "JEV_RESULT_TOO_LARGE"} {
		mode.Store(m)
		before := calls.Load()
		r, err = reg.ExecuteModel(t.Context(), name, args)
		if err != nil || r.ReasonCode != code || calls.Load() != before+1 {
			t.Fatalf("mode %d: %+v %v calls %d", m, r, err, calls.Load()-before)
		}
	}
	before := calls.Load()
	_, _ = reg.ExecuteModel(t.Context(), name, map[string]interface{}{"state": "x", "questions": map[string]interface{}{"bad": map[string]interface{}{"type": "chat", "instructions": "write"}}})
	if calls.Load() != before {
		t.Fatal("invalid request reached network")
	}
	reg.SetSafeSource(func() (string, bool) { return "fixture SAFE", true })
	r, err = reg.ExecuteModel(t.Context(), name, args)
	if err != nil || r.Error == "" || calls.Load() != before {
		t.Fatalf("SAFE: %+v %v", r, err)
	}
	reg.SetSafeSource(nil)
	h.ReplacePolicy(map[string]broker.Grant{}, profiles, nil)
	r, err = reg.ExecuteModel(t.Context(), name, args)
	if err != nil || r.Error == "" || calls.Load() != before {
		t.Fatalf("ungranted: %+v %v", r, err)
	}
	h.ReplacePolicy(grant, profiles, nil)
	mode.Store(5)
	ctx, cancel := context.WithCancel(t.Context())
	cancel()
	r, err = reg.ExecuteModel(ctx, name, args)
	if calls.Load() != before || err == nil && r.Error == "" {
		t.Fatalf("cancelled dispatch: %+v %v", r, err)
	}
	ctx, cancel = context.WithCancel(t.Context())
	done := make(chan struct{})
	go func() { defer close(done); _, _ = reg.ExecuteModel(ctx, name, args) }()
	select {
	case <-entered:
	case <-time.After(5 * time.Second):
		cancel()
		t.Fatal("request never reached transport")
	}
	cancel()
	select {
	case <-done:
	case <-time.After(5 * time.Second):
		t.Fatal("in-flight cancellation did not return")
	}
	if calls.Load() != before+1 {
		t.Fatal("cancelled request was retried")
	}
	recs, e := st.PluginReceipts(meta.ID)
	if e != nil {
		t.Fatal(e)
	}
	if len(recs) == 0 {
		t.Fatal("no broker receipts")
	}
	for _, rec := range recs {
		if strings.Contains(fmt.Sprint(rec), secret) {
			t.Fatal("credential in receipt")
		}
	}
	t.Logf("real WASM, T2 fixture, broker, mixed answers, typed refusals, SAFE, grants, cancellation; %d HTTP requests", calls.Load())
}
