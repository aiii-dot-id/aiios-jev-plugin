func (f jevTransport) RoundTrip(r *http.Request) (*http.Response, error) { return f(r) }

func jevStaged(t *testing.T) (packagetest.PackageSpec, string) {
	t.Helper()
	root := os.Getenv("JEV_STAGE")
	if root == "" {
		t.Fatal("JEV_STAGE is required")
	}
	manifest, err := os.ReadFile(filepath.Join(root, "manifest.json"))
	if err != nil {
		t.Fatal(err)
	}
	var meta struct {
		ID, Version  string
		Capabilities []string `json:"capability_envelope"`
	}
	if err = json.Unmarshal(manifest, &meta); err != nil {
		t.Fatal(err)
	}
	spec := packagetest.PackageSpec{Root: meta.ID + "-" + meta.Version, Manifest: manifest, InstallFiles: map[string][]byte{}}
	err = filepath.WalkDir(filepath.Join(root, "install-root"), func(p string, d os.DirEntry, e error) error {
		if e != nil || d.IsDir() {
			return e
		}
		rel, e := filepath.Rel(filepath.Join(root, "install-root"), p)
		if e != nil {
			return e
		}
		b, e := os.ReadFile(p)
		spec.InstallFiles[filepath.ToSlash(rel)] = b
		return e
	})
	if err != nil {
		t.Fatal(err)
	}
	if err = signer(t).SignT2(&spec, meta.ID, meta.Version, meta.Capabilities); err != nil {
		t.Fatal(err)
	}
	return spec, meta.ID
}

type jevHost struct {
	reg    *tools.Registry
	h      *broker.Host
	ap     *ActivePlugin
	id     string
	values map[string]interface{}
	grant  broker.Grant
}

func jevActivate(t *testing.T, transport http.RoundTripper, sandbox broker.Sandbox, files bool, values map[string]interface{}) *jevHost {
	t.Helper()
	spec, id := jevStaged(t)
	grant := broker.Grant{Hosts: []string{"api.typesafe.ai:443"}, CredentialHandles: []string{"jev"}, Files: files}
	h, err := broker.New(broker.Config{Store: newBrokerStore(t), Grants: map[string]broker.Grant{id: grant}, AuthProfiles: jevProfiles, Guard: func(context.Context, string) error { return nil }, Transport: transport, Sandbox: sandbox})
	if err != nil {
		t.Fatal(err)
	}
	reg := newRegistry(t)
	ap, err := Activate(t.Context(), writePkg(t, spec), reg, &Options{Roots: signerRoots(t), Broker: h, Settings: func(string) map[string]interface{} { return values }})
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = ap.Deactivate(context.Background()) })
	return &jevHost{reg: reg, h: h, ap: ap, id: id, values: values, grant: grant}
}

func (j *jevHost) ask(t *testing.T, ctx context.Context, args string) (tools.Result, map[string]json.RawMessage) {
	t.Helper()
	var a map[string]interface{}
	if err := json.Unmarshal([]byte(args), &a); err != nil {
		t.Fatal(err)
	}
	r, err := j.reg.ExecuteModel(ctx, ToolNameFor(j.id, "ask"), a)
	if err != nil {
		t.Fatalf("ask: %v", err)
	}
	if r.Error != "" {
		return r, nil
	}
	_, content, ok := untrusted.Split(r.Output)
	if !ok {
		t.Fatalf("the result is not wrapped as untrusted: %.120q", r.Output)
	}
	var m map[string]json.RawMessage
	if err := json.Unmarshal([]byte(content), &m); err != nil {
		t.Fatalf("the reply is not JSON: %v %.200q", err, content)
	}
	if string(m["judge"]) != `"api.typesafe.ai"` || strings.Contains(content, jevSecret) {
		t.Fatalf("the reply names no judge, or carries the key: %.200s", content)
	}
	return r, m
}

type jevJudge struct {
	mu      sync.Mutex
	bodies  [][]byte
	headers []http.Header
}

func (j *jevJudge) sent() ([][]byte, []http.Header) {
	j.mu.Lock()
	defer j.mu.Unlock()
	return append([][]byte(nil), j.bodies...), append([]http.Header(nil), j.headers...)
}

func wrote(r *http.Request) []byte {
	data, _ := io.ReadAll(r.Body)
	if trace := httptrace.ContextClientTrace(r.Context()); trace != nil && trace.WroteRequest != nil {
		trace.WroteRequest(httptrace.WroteRequestInfo{})
	}
	return data
}

func (j *jevJudge) answer(t *testing.T, r *http.Request) (*http.Response, error) {
	data := wrote(r)
	j.mu.Lock()
	j.bodies, j.headers = append(j.bodies, data), append(j.headers, r.Header.Clone())
	j.mu.Unlock()
	if r.URL.Host != "api.typesafe.ai" || r.Header.Get("Authorization") != "Bearer "+jevSecret || r.Method != "POST" || r.URL.Path != "/v1/systemone" {
		t.Errorf("wrong request: %s %s", r.Method, r.URL)
	}
	var req struct {
		Questions map[string]struct {
			Type         string
			Instructions json.RawMessage
			Criteria     json.RawMessage
		}
	}
	if err := json.Unmarshal(data, &req); err != nil {
		return jevReply(422, `{"detail":"not JSON"}`), nil
	}
	answers := map[string]any{}
	for name, q := range req.Questions {
		switch q.Type {
		case "noul":
			var in struct{ Item json.RawMessage }
			judged := q.Instructions
			if json.Unmarshal(q.Instructions, &in) == nil && in.Item != nil {
				judged = in.Item
			}
			p := 0.1
			if bytes.Contains(judged, []byte("MATCH")) {
				p = 0.9
			}
			answers[name] = map[string]any{"type": "noul", "noul": p}
		case "box":
			answers[name] = map[string]any{"type": "box", "box": map[string]float64{"top": 0.1, "bottom": 0.9, "left": 0.2, "right": 0.8}, "confidence": 0.7}
		case "choice":
			var c map[string]json.RawMessage
			json.Unmarshal(q.Criteria, &c)
			names := []string{}
			for k := range c {
				names = append(names, k)
			}
			sort.Strings(names)
			probs := map[string]float64{}
			for i, k := range names {
				probs[k] = map[bool]float64{true: 1}[i == 0]
			}
			answers[name] = map[string]any{"type": "choice", "choice": names[0], "probabilities": probs, "confidence": 0.9}
		case "score":
			var levels []string
			json.Unmarshal(q.Criteria, &levels)
			probs, legend := map[string]float64{}, map[string]string{}
			for i, l := range levels {
				probs[fmt.Sprint(i)], legend[fmt.Sprint(i)] = map[bool]float64{true: 1}[i == 0], l
			}
			answers[name] = map[string]any{"type": "score", "score": 0, "probabilities": probs, "legend": legend, "confidence": 0.9}
		}
	}
	b, _ := json.Marshal(map[string]any{"model": "jev-1.14.0", "answers": answers, "usage": map[string]int{"input_tokens": len(data) / 3, "output_tokens": 9}})
	return jevReply(200, string(b)), nil
}

func jevReply(status int, body string) *http.Response {
	return &http.Response{StatusCode: status, Header: http.Header{"Content-Type": []string{"application/json"}}, Body: io.NopCloser(strings.NewReader(body))}
}

func sum16(b []byte) string { s := sha256.Sum256(b); return hex.EncodeToString(s[:8]) }

func pngOf(w, h int) []byte {
	b := []byte("\x89PNG\r\n\x1a\n\x00\x00\x00\x0dIHDR")
	b = append(b, byte(w>>24), byte(w>>16), byte(w>>8), byte(w), byte(h>>24), byte(h>>16), byte(h>>8), byte(h), 8, 6, 0, 0, 0, 0, 0, 0, 0)
	return append(b, bytes.Repeat([]byte{0x5A}, 300)...)
}

func jpegOf(w, h, orient int) []byte {
	b := []byte{0xFF, 0xD8}
	x := []byte("Exif\x00\x00II\x2a\x00\x08\x00\x00\x00\x01\x00\x12\x01\x03\x00\x01\x00\x00\x00")
	x = append(x, byte(orient), 0, 0, 0, 0, 0, 0, 0)
	b = append(append(b, 0xFF, 0xE1, byte((len(x)+2)>>8), byte(len(x)+2)), x...)
	b = append(b, 0xFF, 0xC0, 0, 17, 8, byte(h>>8), byte(h), byte(w>>8), byte(w), 3, 1, 0x22, 0, 2, 0x11, 1, 3, 0x11, 1)
	return append(append(b, bytes.Repeat([]byte{0x5A}, 300)...), 0xFF, 0xD9)
}

func jpegWithIFD(w, h int, ifd uint32) []byte {
	x := []byte("Exif\x00\x00II\x2a\x00")
	x = append(x, byte(ifd), byte(ifd>>8), byte(ifd>>16), byte(ifd>>24))
	x = append(x, bytes.Repeat([]byte{0}, 16)...)
	b := append([]byte{0xFF, 0xD8, 0xFF, 0xE1, byte((len(x) + 2) >> 8), byte(len(x) + 2)}, x...)
	b = append(b, 0xFF, 0xC0, 0, 17, 8, byte(h>>8), byte(h), byte(w>>8), byte(w), 3, 1, 0x22, 0, 2, 0x11, 1, 3, 0x11, 1)
	return append(append(b, bytes.Repeat([]byte{0x5A}, 300)...), 0xFF, 0xD9)
}

func TestGenericJevRealBroker(t *testing.T) {
	t.Setenv("JEV_FIXTURE_SECRET", jevSecret)
	var calls atomic.Int32
	var mode atomic.Int32
	entered := make(chan struct{}, 1)
	judge := &jevJudge{}
	transport := jevTransport(func(r *http.Request) (*http.Response, error) {
		calls.Add(1)
		if r.URL.Host != "api.typesafe.ai" || r.Header.Get("Authorization") != "Bearer "+jevSecret {
			t.Error("wrong destination or missing broker credential")
		}
		if r.URL.Path == "/v1/models" {
			if mode.Load() == 4 {
				return jevReply(200, `{"models":[{"name":"jev-latest"},{}]}`), nil
			}
			return jevReply(200, `{"models":[{"name":"jev-latest","description":"General-purpose system one model.","release_date":"2026-09-15"},{"name":"jev-preview"}]}`), nil
		}
		switch mode.Load() {
		case 1:
			wrote(r)
			return jevReply(200, `{"model":"jev-1.14.0","answers":{"n":{"type":"noul","noul":2}},"usage":{"input_tokens":1,"output_tokens":1}}`), nil
		case 2:
			wrote(r)
			return jevReply(429, `{"error":"busy"}`), nil
		case 5:
			entered <- struct{}{}
			<-r.Context().Done()
			return nil, r.Context().Err()
		case 6:
			wrote(r)
			return jevReply(400, `{"detail":"max_tokens_exceeded: the request is longer than the model's context"}`), nil
		}
		return judge.answer(t, r)
	})
	j := jevActivate(t, transport, nil, false, map[string]interface{}{"api_key": "jev", "model": "invalid model"})
	foundChoices := false
	for _, setting := range j.ap.Settings {
		if setting.Key == "model" && setting.ChoicesFrom == "models" {
			foundChoices = true
		}
	}
	if !foundChoices || len(j.reg.Search("jev", 10)) == 0 {
		t.Fatal("packaging lost the model picker, or the plugin is not discoverable")
	}
	models := ToolNameFor(j.id, "models")
	r, err := j.reg.ExecuteModel(t.Context(), models, map[string]interface{}{})
	if err != nil || r.Error != "" || !strings.Contains(r.Output, `"judge":"api.typesafe.ai"`) || !strings.Contains(r.Output, `{"value":"jev-latest","label":"2026-09-15 General-purpose system one model."}`) {
		t.Fatalf("models: %+v %v", r, err)
	}
	mode.Store(4)
	if r, err = j.reg.ExecuteModel(t.Context(), models, map[string]interface{}{}); err != nil || r.Error == "" {
		t.Fatalf("partial model list accepted: %+v %v", r, err)
	}
	mode.Store(0)
	j.values["model"] = "jev-latest"
	name := ToolNameFor(j.id, "ask")
	card, err := j.reg.Show(name)
	if err != nil || card.Effects != "write.external" {
		t.Fatalf("descriptor: %+v %v", card, err)
	}

	if want := jevStagedMaximum(t, "max_reply_bytes"); card.MaxResultBytes != want {
		t.Fatalf("ask declares %d bytes, the reply budget allows %d", card.MaxResultBytes, want)
	}
	if mcard, err := j.reg.Show(models); err != nil || mcard.MaxResultBytes < len(`{"choices":[]}`)+256*(len(`{"value":""},`)+128) {
		t.Fatalf("models declares %+v %v", mcard, err)
	}
	args := `{"state":"evidence","questions":[{"id":"n","instructions":"Is it supported?"},{"id":"c","type":"choice","instructions":"Choose","criteria":{"a":null,"b":null}},{"id":"s","type":"score","instructions":"Rate","criteria":["none","some","all"]}]}`
	r, m := j.ask(t, t.Context(), args)
	if m == nil || !strings.Contains(string(m["answers"]), `"n":{"noul":0.1}`) || !strings.Contains(string(m["answers"]), `"s":{"score":0,"probabilities":{"0":1,"1":0,"2":0},"confidence":0.9}`) {
		t.Fatalf("mixed reply: %+v", r)
	}
	bodies, _ := judge.sent()
	if b := bodies[len(bodies)-1]; !bytes.Contains(b, []byte(`"n":{"type":"noul","instructions":"Is it supported?"}`)) || bytes.Contains(b, []byte("_host")) || !bytes.HasPrefix(b, []byte(`{"model":"jev-latest","state":"evidence","questions":{`)) {
		t.Fatalf("outbound: %s", b)
	}

	before := calls.Load()
	if r, _ = j.ask(t, t.Context(), `{"state":{"$file":"notes.md"},"questions":[{"id":"q","instructions":"Is it?"}]}`); r.Error == "" || r.ReasonCode == "" || calls.Load() != before {
		t.Fatalf("a reference without the files grant: %+v", r)
	}
	t.Logf("a reference without the files grant: %s: %s", r.ReasonCode, r.Error)

	j.values["timeout_ms"] = jevStagedMaximum(t, "timeout_ms")
	if r, _ = j.ask(t, t.Context(), args); r.Error != "" {
		t.Fatalf("timeout_ms at its declared maximum: %+v", r)
	}
	delete(j.values, "timeout_ms")
	said := map[int32]string{2: `{"error":"busy"}`, 6: `"max_tokens_exceeded: the request is longer than the model's context"`}
	for md, code := range map[int32]string{1: "JEV_RESPONSE_INVALID", 2: "JEV_RATE_LIMITED", 6: "JEV_REQUEST_REFUSED"} {
		mode.Store(md)
		before := calls.Load()
		a := args
		if md == 1 {
			a = `{"state":"x","questions":[{"id":"n","instructions":"Is it?"}]}`
		}
		r, _ = j.ask(t, t.Context(), a)
		if r.ReasonCode != code || calls.Load() != before+1 {
			t.Fatalf("mode %d: %+v calls %d", md, r, calls.Load()-before)
		}
		if words := said[md]; words != "" && !strings.Contains(r.Error, "the judge said: "+words) {
			t.Fatalf("mode %d lost the judge's words: %q", md, r.Error)
		}
		if md == 6 && !strings.Contains(r.Error, "long_context true") {
			t.Fatalf("too long gave no advice: %q", r.Error)
		}
	}
	mode.Store(0)
	before = calls.Load()
	_, _ = j.reg.ExecuteModel(t.Context(), name, map[string]interface{}{"state": "x", "questions": []interface{}{map[string]interface{}{"id": "bad", "type": "chat", "instructions": "write"}}})
	if calls.Load() != before {
		t.Fatal("invalid request reached network")
	}

	for _, bad := range []string{`{"state":"x","questions":[{"id":"q","type":"chat","instructions":"x"}]}`, `{"state":"x","questions":[{"id":"q","instructions":"x","extra":1}]}`, `{"state":"x","questions":[{"id":"q","instructions":"x"},{"id":"q","instructions":"y"}]}`} {
		if r, _ = j.ask(t, t.Context(), bad); r.Error == "" || calls.Load() != before {
			t.Fatalf("%s: %+v", bad, r)
		}
		t.Logf("refused before the call: %s: %s", r.ReasonCode, r.Error)
	}
	plain := map[string]interface{}{"state": "x", "questions": []interface{}{map[string]interface{}{"id": "q", "instructions": "Is it?"}}}
	j.reg.SetSafeSource(func() (string, bool) { return "fixture SAFE", true })
	if r, err = j.reg.ExecuteModel(t.Context(), name, plain); err != nil || r.Error == "" || calls.Load() != before {
		t.Fatalf("SAFE: %+v %v", r, err)
	}
	j.reg.SetSafeSource(nil)
	j.h.ReplacePolicy(map[string]broker.Grant{}, jevProfiles, nil)
	if r, _ = j.ask(t, t.Context(), args); r.Error == "" || calls.Load() != before {
		t.Fatalf("ungranted: %+v", r)
	}
	j.h.ReplacePolicy(map[string]broker.Grant{j.id: j.grant}, jevProfiles, nil)
	mode.Store(5)
	ctx, cancel := context.WithCancel(t.Context())
	cancel()
	if r, err = j.reg.ExecuteModel(ctx, name, plain); calls.Load() != before || err == nil && r.Error == "" {
		t.Fatalf("cancelled dispatch: %+v %v", r, err)
	}
	ctx, cancel = context.WithCancel(t.Context())
	done := make(chan struct{})
	go func() { defer close(done); _, _ = j.reg.ExecuteModel(ctx, name, plain) }()
	select {
	case <-entered:
	case <-time.After(5 * time.Second):
		cancel()
		t.Fatal("request never reached transport")
	}
	cancel()
	select {
	case <-done:
	case <-time.After(5 * time.Second):
		t.Fatal("in-flight cancellation did not return")
	}
	if calls.Load() != before+1 {
		t.Fatal("cancelled request was retried")
	}
	t.Logf("real WASM, T2 fixture, broker, mixed answers, typed refusals, SAFE, grants, cancellation; %d HTTP requests", calls.Load())
}

	}
	for i := 0; i < 70; i++ {
		files[fmt.Sprintf("inbox/%04d.eml", i)] = []byte(fmt.Sprintf("ticket %d: %s", i, map[bool]string{true: "MATCH refund", false: "a question"}[i%3 == 0]))
	}

	policy := firewall.DefaultPolicy()
	policy.SetPlaces([]*firewall.Rule{{ID: "sub.data", Kind: firewall.KindSubstrate, Path: filepath.Join(home, "data"), Reason: "the host keeps it", Enforced: true}})

}

func TestGenericJevReferencesThroughTheBroker(t *testing.T) {
	t.Setenv("JEV_FIXTURE_SECRET", jevSecret)
	home, world := jevWorld(t)
	judge := &jevJudge{}
	j := jevActivate(t, jevTransport(func(r *http.Request) (*http.Response, error) { return judge.answer(t, r) }), world, true, map[string]interface{}{"api_key": "jev"})

	r, m := j.ask(t, t.Context(), `{"state":{"rule":{"$file":"notes/policy.md"},"photo":{"$file":"photos/kitchen.jpg"}},"questions":[{"id":"covered","instructions":"Does `+"`rule`"+` cover refunds?"},{"id":"mug","type":"box","instructions":"the mug in `+"`photo`"+`"}]}`)
	if m == nil {
		t.Fatalf("one state with references: %+v", r)
	}
	bodies, headers := judge.sent()
	var req struct {
		State  map[string]string
		Assets map[string]struct{ Type, Format, Data_base64 string }
	}
	if err := json.Unmarshal(bodies[0], &req); err != nil {
		t.Fatal(err)
	}
	photo, _ := os.ReadFile(filepath.Join(home, "photos/kitchen.jpg"))
	note, _ := os.ReadFile(filepath.Join(home, "notes/policy.md"))
	if req.State["rule"] != string(note) || req.Assets[req.State["photo"]].Data_base64 != base64.StdEncoding.EncodeToString(photo) || headers[0].Get("TypeSafe-Danger-Zone-Images") != "true" {
		t.Fatalf("the request: %.300s headers %v", bodies[0], headers[0])
	}
	var sent map[string]string
	if json.Unmarshal(m["sent"], &sent); len(sent) != 2 || sent["notes/policy.md"] != sum16(note) || sent["photos/kitchen.jpg"] != sum16(photo) || !strings.Contains(string(m["orientation"]), "EXIF 6") {
		t.Fatalf("sent %s, orientation %s", m["sent"], m["orientation"])
	}

	seen := map[string]int{}
	from, calls := "", 0
	for {
		args := `{"state":"A ticket matters when it asks for a refund.","questions":[{"id":"refund","instructions":"Does ` + "`item`" + ` say MATCH?"}],"items":{"$files":"inbox/*.eml"},"rank":"refund"`
		if from != "" {
			args += `,"from":` + from
		}
		r, m = j.ask(t, t.Context(), args+"}")
		if m == nil {
			t.Fatalf("call %d: %+v", calls, r)
		}
		calls++
		var rows []struct{ Ref, Refused string }
		json.Unmarshal(m["rows"], &rows)
		for _, row := range rows {
			seen[row.Ref]++
		}
		if string(m["next"]) == "null" || calls > 20 {
			break
		}
		from = string(m["next"])
	}
	if len(seen) != 70 || seen["inbox/.hidden.eml"] != 0 || seen["inbox/b.txt"] != 0 {
		t.Fatalf("%d items in %d calls: %v", len(seen), calls, seen)
	}
	for p, n := range seen {
		if n != 1 {
			t.Fatalf("%s answered %d times", p, n)
		}
	}

	for _, off := range []bool{false, true} {
		world.SetSandboxOff(off)
		for _, p := range []string{"data/credentials/jev.key", filepath.Join(home, "data", "config.json")} {
			before, _ := judge.sent()
			r, _ = j.ask(t, t.Context(), `{"state":{"$file":"`+p+`"},"questions":[{"id":"q","instructions":"Is it?"}]}`)
			after, _ := judge.sent()
			if r.Error == "" || len(after) != len(before) || strings.Contains(r.Output+r.Error, "SECRET") {
				t.Fatalf("the data directory, sandbox off %v: %+v", off, r)
			}
			t.Logf("sandbox off %v, %s: %s", off, p, r.ReasonCode)
		}
	}
	world.SetSandboxOff(false)

	bodies, _ = judge.sent()
	before := len(bodies)
	from, calls = "", 0
	for {
		args := `{"state":{"rule":{"$file":"rule.md"},"reference":{"$file":"ref.png"}},"questions":[{"id":"same","instructions":"Is the mug in ` + "`item`" + ` the one in ` + "`reference`" + `?"},{"id":"where","type":"box","instructions":"the mug in ` + "`item`" + `"}],"items":{"$files":"img/*"}`
		if from != "" {
			args += `,"from":` + from
		}
		r, m = j.ask(t, t.Context(), args+"}")
		if m == nil {
			t.Fatalf("image items: %+v", r)
		}
		calls++
		if string(m["next"]) == "null" || calls > 5 {
			break
		}
		from = string(m["next"])
	}
	bodies, headers = judge.sent()
	if calls != 2 || len(bodies) != before+2 {
		t.Fatalf("%d calls, %d requests for two image items", calls, len(bodies)-before)
	}
	for i, b := range bodies[before:] {
		var q struct {
			State     map[string]string
			Questions map[string]struct{ Instructions struct{ Item string } }
			Assets    map[string]json.RawMessage
		}
		json.Unmarshal(b, &q)
		if q.State["rule"] != "A mug counts when its handle shows." || len(q.Assets) != 2 || headers[before+i].Get("TypeSafe-Danger-Zone-Images") != "true" {
			t.Fatalf("image request %d: %.300s", i, b)
		}
		for _, qq := range q.Questions {
			if _, ok := q.Assets[qq.Instructions.Item]; !ok || qq.Instructions.Item == q.State["reference"] {
				t.Fatalf("image request %d: the item is not in its place", i)
			}
		}
	}
}

type slowGate struct {
	tools.Gate
	path  string
	delay time.Duration
}

func (g slowGate) Admit(op, path string) (string, string, error) {
	if path == g.path {
		time.Sleep(g.delay)
	}
	return g.Gate.Admit(op, path)
}

type slowWorld struct {
	*tools.Registry
	path  string
	delay time.Duration
}

func (w slowWorld) Look() tools.Gate { return slowGate{w.Registry.Look(), w.path, w.delay} }

func TestGenericJevTheAttemptFitsTheInvocation(t *testing.T) {
	t.Setenv("JEV_FIXTURE_SECRET", jevSecret)
	for _, tc := range []struct {
		name  string
		delay time.Duration
		judge func(*http.Request) (*http.Response, error)
		code  string
		says  string
	}{
		{"not written, then the deadline", 30 * time.Second, func(r *http.Request) (*http.Response, error) {
			<-r.Context().Done()
			return nil, r.Context().Err()
		}, "JEV_TRANSPORT_NOT_SENT", "Not sent"},
		{"too long at the last moment", 30 * time.Second, func(r *http.Request) (*http.Response, error) {
			wrote(r)
			deadline, _ := r.Context().Deadline()
			time.Sleep(time.Until(deadline) - 2*time.Second)
			return jevReply(400, `{"detail":"max_tokens_exceeded"}`), nil
		}, "JEV_REQUEST_REFUSED", "long_context true"},
		{"out of time", 86 * time.Second, nil, "JEV_NOT_SENT_OUT_OF_TIME", "Not sent: out of time"},
	} {
		t.Run(tc.name, func(t *testing.T) {
			t.Parallel()
			_, world := jevWorld(t)
			var posts atomic.Int32
			transport := jevTransport(func(r *http.Request) (*http.Response, error) {
				posts.Add(1)
				return tc.judge(r)
			})
			j := jevActivate(t, transport, slowWorld{world, "rule.md", tc.delay}, true, map[string]interface{}{"api_key": "jev", "timeout_ms": 60000})
			start := time.Now()
			r, _ := j.ask(t, t.Context(), `{"state":{"$file":"rule.md"},"questions":[{"id":"q","instructions":"Does `+"`state`"+` name a mug?"}]}`)
			took := time.Since(start)
			t.Logf("%s after %v: %s: %s", tc.name, took.Round(time.Millisecond), r.ReasonCode, r.Error)
			if r.ReasonCode != tc.code || !strings.Contains(r.Error, tc.says) || took > 88*time.Second {
				t.Fatalf("%s after %v: %+v", tc.name, took, r)
			}
			if tc.judge == nil && posts.Load() != 0 || tc.judge != nil && posts.Load() != 1 {
				t.Fatalf("%d requests", posts.Load())
			}
		})
	}
}

func TestGenericJevALostAnswerIsUnknownAndNotAskedAgain(t *testing.T) {
	t.Setenv("JEV_FIXTURE_SECRET", jevSecret)
	judge := &jevJudge{}
	var lose atomic.Bool
	var posts atomic.Int32
	var mu sync.Mutex
	var lostBodies [][]byte
	transport := jevTransport(func(r *http.Request) (*http.Response, error) {
		posts.Add(1)
		if !lose.Load() {
			return judge.answer(t, r)
		}
		data := wrote(r)
		mu.Lock()
		lostBodies = append(lostBodies, data)
		mu.Unlock()
		<-r.Context().Done()
		return nil, r.Context().Err()
	})
	j := jevActivate(t, transport, nil, false, map[string]interface{}{"api_key": "jev", "timeout_ms": 3000})
	call := func(args string) (tools.Result, map[string]json.RawMessage, int32, time.Duration) {
		before, start := posts.Load(), time.Now()
		r, m := j.ask(t, t.Context(), args)
		return r, m, posts.Load() - before, time.Since(start)
	}
	lost := func(what string, r tools.Result, n int32, took time.Duration, again string) {
		t.Helper()
		t.Logf("%s after %v, %d POST: %s: %s", what, took.Round(time.Millisecond), n, r.ReasonCode, r.Error)
		if r.ReasonCode != "JEV_TRANSPORT_UNKNOWN" || !strings.Contains(r.Error, "no answer came back: the answers are unknown and may have been charged") ||
			!strings.Contains(r.Error, again) || n != 1 || took > 20*time.Second {
			t.Fatalf("%s: %+v; %d POST in %v", what, r, n, took)
		}
	}

	lose.Store(true)
	r, _, n, took := call(`{"state":"x","questions":[{"id":"q","instructions":"Is it?"}]}`)
	lost("one state", r, n, took, "asking again asks again, and may be charged again")

	items := `{"state":"x","questions":[{"id":"q","instructions":"Does ` + "`item`" + ` say MATCH?"}],"items":["A` +
		strings.Repeat("a", 80_000) + `","B` + strings.Repeat("b", 80_000) + `"]`
	lose.Store(false)
	r, m, n, _ := call(items + "}")
	if m == nil || n != 1 || !strings.Contains(string(m["next"]), `"after":0,`) {
		t.Fatalf("the first item: %+v; next %s; %d POST", r, m["next"], n)
	}
	from := string(m["next"])
	lose.Store(true)
	for _, what := range []string{"many items", "a second call with the same from"} {
		r, _, n, took = call(items + `,"from":` + from + "}")
		lost(what, r, n, took, "nothing in this call was consumed: the same from asks these items again")
		mu.Lock()
		asked := lostBodies[len(lostBodies)-1]
		mu.Unlock()
		if !bytes.Contains(asked, []byte(`"item":"B`)) || bytes.Contains(asked, []byte(`"item":"A`)) {
			t.Fatalf("%s asked another item: %.160s", what, asked)
		}
	}
	mu.Lock()
	defer mu.Unlock()
	if len(lostBodies) != 3 || posts.Load() != 4 {
		t.Fatalf("%d lost requests, %d POSTs in all; want 3 and 4", len(lostBodies), posts.Load())
	}
}

func jevStagedMaximum(t *testing.T, key string) int {
	t.Helper()
	b, err := os.ReadFile(filepath.Join(os.Getenv("JEV_STAGE"), "install-root", "settings.json"))
	if err != nil {
		t.Fatal(err)
	}
	var decl []struct {
		Key     string   `json:"key"`
		Maximum *float64 `json:"maximum"`
	}
	if err := json.Unmarshal(b, &decl); err != nil {
		t.Fatal(err)
	}
	for _, d := range decl {
		if d.Key == key && d.Maximum != nil {
			return int(*d.Maximum)
		}
	}
	t.Fatalf("the staged settings declare no maximum for %s", key)
	return 0
}

func TestGenericJevAMalformedEXIFOffsetIsReadNotTrapped(t *testing.T) {

	home, world := jevWorld(t)
	judge := &jevJudge{}

	}
}
