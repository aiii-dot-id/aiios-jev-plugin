func TestGenericJevDeliveryBound(t *testing.T) {

	for _, c := range []struct{ size, limit int }{{16 << 10, 20000}, {jevStagedMaximum(t, "max_reply_bytes"), limits.DefaultToolResultChars}} {
		prefix, suffix := `{"answers":{},"padding":"`, `"}`
		raw := prefix + strings.Repeat("x", c.size-len(prefix)-len(suffix)) + suffix
		full := untrusted.Wrap("plugin id.aiii.jev", raw)
		if !jevDelivered(t, full, c.limit, full) {
			t.Fatalf("a %d-byte Jev result was not delivered whole at %d", c.size, c.limit)
		}
	}
}

func TestGenericJevCursorSurvivesASmallWindow(t *testing.T) {
	var rows []string
	for i := 0; i < 120; i++ {
		rows = append(rows, fmt.Sprintf(`{"ref":"inbox/%04d.eml","sha256":"0c6e81f2a4b9d750","relevant":{"noul":0.98}}`, i))
	}
	next := `"next":{"after":"inbox/0119.eml","model":"jev-1.14.0","screen":"` + strings.Repeat("9c1e", 16) + `"}`
	raw := `{"judge":"api.typesafe.ai","model":"jev-1.14.0","rows":[` + strings.Join(rows, ",") + `],"usage":{"input_tokens":98210,"output_tokens":410},"visited":120,` + next + `}`
	if !json.Valid([]byte(raw)) || len(raw) < 8000 {
		t.Fatalf("the fixture: %d bytes", len(raw))
	}
	full := untrusted.Wrap("plugin id.aiii.jev", raw)
	if !jevDelivered(t, full, 4000, next) {
		t.Fatal("the cursor did not survive a 4,000-character window")
	}
	if jevDelivered(t, full, 4000, rows[0]) {
		t.Fatal("the window held the whole reply; the cut was not exercised")
	}
}

func jevDelivered(t *testing.T, result string, limit int, want string) bool {
	t.Helper()
	client := &scriptLLM{script: []llm.Response{resp("", toolCall("j1", "pl_id_aiii_jev_ask", `{}`)), textResp("done")}}
	tr := &fakeTranscript{limit: 20000}
	loop := New(client, &fakeTools{results: map[string]string{"pl_id_aiii_jev_ask": result}}, &fakeDefs{}, tr, nil, testBounds(Config{MaxToolResultChars: limit}))
	if _, err := loop.Run(t.Context(), "system", []llm.Message{{Role: "user", Content: "evaluate"}}); err != nil {
		t.Fatal(err)
	}
	for _, m := range client.requests[1] {
		if strings.Contains(m.Content, want) {
			return true
		}
	}
	return false
}

func jevStagedMaximum(t *testing.T, key string) int {
	t.Helper()
	b, err := os.ReadFile(filepath.Join(os.Getenv("JEV_STAGE"), "install-root", "settings.json"))
	if err != nil {
		t.Fatal(err)
	}
	var decl []struct {
		Key     string   `json:"key"`
		Maximum *float64 `json:"maximum"`
	}
	if err := json.Unmarshal(b, &decl); err != nil {
		t.Fatal(err)
	}
	for _, d := range decl {
		if d.Key == key && d.Maximum != nil {
			return int(*d.Maximum)
		}
	}
	t.Fatalf("the staged settings declare no maximum for %s", key)
	return 0
}
